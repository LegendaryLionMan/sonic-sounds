```sql
-- 1. ARTIST — first-class identity, survives across albums
CREATE TABLE artist (
  id                    TEXT PRIMARY KEY,           -- ULID
  name                  TEXT NOT NULL UNIQUE,
  type                  TEXT NOT NULL,              -- 'solo' | 'band'
  bio_short             TEXT,
  bio_medium            TEXT,
  bio_long              TEXT,
  voice_characteristic  TEXT NOT NULL,              -- default --vocals descriptor
  genre_primary         TEXT NOT NULL,
  genre_secondary       TEXT,
  reference_artists     TEXT,                       -- JSON array of artist names
  palette_house         TEXT,                       -- JSON array of 5 hex codes
  motif_house           TEXT,                       -- artist's recurring image
  typography_json       TEXT,                       -- {display, body, mono}
  artist_photo_path     TEXT,
  distributor           TEXT,
  distributor_artist_id TEXT,
  label                 TEXT,
  copyright_holder      TEXT,
  composer_credit       TEXT,
  lyricist_credit       TEXT,
  isrc_registrant_code  TEXT,
  created_at            INTEGER NOT NULL,
  updated_at            INTEGER NOT NULL
);

-- 2. ARTIST_MEMBER — band membership
CREATE TABLE artist_member (
  artist_id             TEXT NOT NULL REFERENCES artist(id) ON DELETE CASCADE,
  member_artist_id      TEXT NOT NULL REFERENCES artist(id) ON DELETE CASCADE,
  role                  TEXT NOT NULL,
  display_order         INTEGER NOT NULL DEFAULT 0,
  PRIMARY KEY (artist_id, member_artist_id, role)
);

-- 3. ALBUM — per-release
CREATE TABLE album (
  id                    TEXT PRIMARY KEY,           -- ULID
  slug                  TEXT NOT NULL UNIQUE,       -- 'maren-sol-half-light-hours'
  artist_id             TEXT NOT NULL REFERENCES artist(id),
  title                 TEXT NOT NULL,
  year                  INTEGER,
  scope                 TEXT,                       -- 'single'|'ep'|'album'|'double-feature'|'concept-piece'
  genre_primary         TEXT,
  genre_secondary       TEXT,
  runtime_target        TEXT,                       -- 'radio-edit'|'standard'|'extended'|'immersive'
  runtime_ms            INTEGER,                    -- SUM of track durations
  palette_json          TEXT,                       -- 5 hex codes
  motif                 TEXT,                       -- this album's recurring image
  concept               TEXT,                       -- narrative arc
  production_style      TEXT,                       -- 'stripped'|'full-band'|'electronic'|'cinematic'|'hybrid'
  vocal_approach        TEXT,                       -- 'solo'|'duet'|'choir'|'spoken-word'|'instrumental'
  language              TEXT,                       -- JSON array
  lyrical_source        TEXT,                       -- 'user-writes'|'agent-writes'|'co-write'
  distribution_intent   TEXT,                       -- 'just-for-me'|'routenote-free'|'full-dsp'|'physical-and-dsp'
  press_scope           TEXT,
  music_videos          TEXT,
  deadline              TEXT,                       -- ISO date
  cassette_sticker_path TEXT,                       -- optional user-uploaded sticker override
  status                TEXT NOT NULL,              -- 'draft'|'intake'|'approved'|'building'|'complete'|'archived'
  brief_json            TEXT,                       -- full 21-topic brief (raw intake JSON)
  directory             TEXT NOT NULL,              -- '~/Documents/Projects/sonic-studio/music/<slug>/'
  created_at            INTEGER NOT NULL,
  updated_at            INTEGER NOT NULL
);
CREATE INDEX idx_album_artist ON album(artist_id);
CREATE INDEX idx_album_status ON album(status);

-- 4. TRACK — one row per .mp3
CREATE TABLE track (
  id                    TEXT PRIMARY KEY,           -- ULID
  album_id              TEXT NOT NULL REFERENCES album(id) ON DELETE CASCADE,
  track_number          INTEGER NOT NULL,           -- 01..NN
  title                 TEXT NOT NULL,
  duration_ms           INTEGER NOT NULL,
  file_path             TEXT NOT NULL,              -- absolute path to .mp3
  file_bytes            INTEGER NOT NULL,
  file_sha256           TEXT NOT NULL UNIQUE,
  mime_type             TEXT NOT NULL DEFAULT 'audio/mpeg',
  isrc                  TEXT,
  bpm                   INTEGER,
  musical_key           TEXT,
  vocal_style_override  TEXT,                       -- per-track vocal variation
  lyrics_path           TEXT,                       -- 'lyrics/NN-<slug>.md'
  lrc_path              TEXT,                       -- 'lyrics-lrc/NN-<slug>.lrc'
  created_at            INTEGER NOT NULL,
  updated_at            INTEGER NOT NULL,
  UNIQUE(album_id, track_number)
);
CREATE INDEX idx_track_album ON track(album_id);

-- 5. PIPELINE_STAGE — 12 layers, one row per (album, layer)
CREATE TABLE pipeline_stage (
  album_id              TEXT NOT NULL REFERENCES album(id) ON DELETE CASCADE,
  stage_number          INTEGER NOT NULL,           -- 01..12
  stage_name            TEXT NOT NULL,
  status                TEXT NOT NULL,              -- 'todo'|'in-progress'|'blocked'|'done'
  artifact_path         TEXT,                       -- path to the produced artifact
  updated_at            INTEGER NOT NULL,
  notes                 TEXT,
  PRIMARY KEY(album_id, stage_number)
);

-- 6. ASSET — sidecar outputs (posters, merch, press, social, video, cover)
CREATE TABLE asset (
  id                    TEXT PRIMARY KEY,
  album_id              TEXT NOT NULL REFERENCES album(id) ON DELETE CASCADE,
  category              TEXT NOT NULL,              -- 'cover'|'poster'|'merch'|'press'|'social'|'video'
  file_path             TEXT NOT NULL,
  mime_type             TEXT NOT NULL,
  file_bytes            INTEGER NOT NULL,
  metadata_json         TEXT,                       -- {aspect_ratio, sku, ...}
  created_at            INTEGER NOT NULL
);
CREATE INDEX idx_asset_album ON asset(album_id);
CREATE INDEX idx_asset_cat ON asset(album_id, category);

-- 7. INTAKE_DRAFT — in-progress brief before it becomes an album
CREATE TABLE intake_draft (
  id                    TEXT PRIMARY KEY,
  artist_id             TEXT REFERENCES artist(id),  -- NULL for blank canvas
  forked_from_album_id  TEXT REFERENCES album(id),   -- NULL for blank canvas
  responses_json        TEXT NOT NULL,                -- raw intake form JSON
  current_topic         INTEGER,                      -- last answered topic (1-21)
  completed             INTEGER NOT NULL DEFAULT 0,   -- 0|1
  created_at            INTEGER NOT NULL,
  updated_at            INTEGER NOT NULL
);

-- 8. CHAT_TUNING — history of brief edits via chat
CREATE TABLE chat_tuning (
  id                    INTEGER PRIMARY KEY AUTOINCREMENT,
  album_id              TEXT NOT NULL REFERENCES album(id) ON DELETE CASCADE,
  role                  TEXT NOT NULL,                -- 'user'|'assistant'
  content               TEXT NOT NULL,
  applied_topics        TEXT,                          -- JSON array of topic IDs affected
  pending_suggestion    TEXT,                          -- JSON of pending topic changes
  occurred_at           INTEGER NOT NULL
);
CREATE INDEX idx_chat_album ON chat_tuning(album_id);
CREATE INDEX idx_chat_time ON chat_tuning(occurred_at);

-- 9. UPLOAD_EVENT — audit log for file uploads (per-album assets)
CREATE TABLE upload_event (
  id                    INTEGER PRIMARY KEY AUTOINCREMENT,
  tmp_path              TEXT NOT NULL,
  final_path            TEXT,
  file_sha256           TEXT,
  size_bytes            INTEGER,
  status                TEXT NOT NULL,                 -- 'started'|'validated'|'moved'|'failed'
  error                 TEXT,
  occurred_at           INTEGER NOT NULL
);
```

### 4.3 Streaming endpoint (the critical one)

```python
# apps/sonic-studio/src/routes/audio.py
#
# Serves `music/<album-slug>/tracks/NN-<slug>.mp3` with HTTP 206 Partial Content
# so the browser's <audio> element can seek without downloading the whole file.
#
# Key requirements:
#  - Accept-Ranges: bytes (always)
#  - HEAD request returns Content-Length + Content-Type
#  - GET with Range header returns 206 + Content-Range
#  - Cache-Control: public, max-age=3600
#  - CORS: allow same-origin (this is a single-port app)
#  - Handle missing file → 404 with album-track context

from fastapi import APIRouter, Request, Response, HTTPException
from fastapi.responses import StreamingResponse
import aiofiles
from pathlib import Path

router = APIRouter()

@router.head("/api/audio/{album_id}/{track_id}")
async def audio_head(album_id: str, track_id: str, db: DbDep):
    track = await db.get_track(track_id)
    if not track or track.album_id != album_id:
        raise HTTPException(404)
    path = Path(track.file_path)
    if not path.is_file():
        raise HTTPException(404)
    size = path.stat().st_size
    return Response(
        headers={
            "Content-Length": str(size),
            "Content-Type": track.mime_type or "audio/mpeg",
            "Accept-Ranges": "bytes",
            "Cache-Control": "public, max-age=3600",
        }
    )

@router.get("/api/audio/{album_id}/{track_id}")
async def audio_get(album_id: str, track_id: str, request: Request, db: DbDep):
    track = await db.get_track(track_id)
    if not track or track.album_id != album_id:
        raise HTTPException(404)
    path = Path(track.file_path)
    if not path.is_file():
        raise HTTPException(404)
    size = path.stat().st_size

    range_header = request.headers.get("range")
    if not range_header:
        # No range — full file
        async def stream_full():
            async with aiofiles.open(path, "rb") as f:
                while chunk := await f.read(64 * 1024):
                    yield chunk
        return StreamingResponse(
            stream_full(),
            media_type=track.mime_type or "audio/mpeg",
            headers={
                "Content-Length": str(size),
                "Accept-Ranges": "bytes",
                "Cache-Control": "public, max-age=3600",
            },
        )

    # Range request — parse "bytes=START-END"
    start, end = parse_range(range_header, size)
    chunk_size = end - start + 1

    async def stream_range():
        async with aiofiles.open(path, "rb") as f:
            await f.seek(start)
            remaining = chunk_size
            while remaining > 0:
                read_size = min(64 * 1024, remaining)
                chunk = await f.read(read_size)
                if not chunk:
                    break
                remaining -= len(chunk)
                yield chunk

    return StreamingResponse(
        stream_range(),
        status_code=206,
        media_type=track.mime_type or "audio/mpeg",
        headers={
            "Content-Range": f"bytes {start}-{end}/{size}",
            "Content-Length": str(chunk_size),
            "Accept-Ranges": "bytes",
            "Cache-Control": "public, max-age=3600",
        },
    )

def parse_range(header: str, total_size: int) -> tuple[int, int]:
    """Parse 'bytes=START-END' or 'bytes=START-' (open-ended)."""
    if not header.startswith("bytes="):
        return 0, total_size - 1
    spec = header[len("bytes="):]
    start_s, _, end_s = spec.partition("-")
    start = int(start_s) if start_s else 0
    end = int(end_s) if end_s else total_size - 1
    end = min(end, total_size - 1)
    return start, end
```

## 8. Pitfalls (from `update-skill-discipline` canon)

| Trap | Mitigation |
|---|---|
| **Don't trust file properties from internal READMEs** | Verify all bitrate/duration/sample-rate via `ffprobe` at endpoint startup |
| **Don't double-instrument** | The studio IS the UI shell; every "generate" button calls the existing CLI. No re-implementation of `embed-id3-metadata.py` / `lyrics-to-lrc.py` |
| **Don't use bash heredoc for ffmpeg** | Use `subprocess.run(args_list, ...)` from `execute_code` (carried from Maren) |
| **Don't burn the video quota** | 1 video generation per visit max; check `mmx_quota_show` first |
| **Don't ship the MP3 without ID3 tags** | Layer 11 runs automatically before Layer 12; pipeline refuses to mark Layer 3 done without Layer 11+12 |
| **Don't lose work to a closed tab** | All forms autosave to `localStorage` + draft endpoints (the existing intake.html already does this) |
| **Don't fake streamed bytes** | The audio endpoint must actually stream the file; verify with `curl --range` + sha256 |
| **Don't trust the page DOM during drag-drop** | Playwright drag-drop tests need real `dragenter`/`dragover`/`drop` events; manual `setAttribute('dragged')` lies |
| **Don't mirror before validating md5** | Per the user's mandate: byte-match verified BEFORE claiming done |
